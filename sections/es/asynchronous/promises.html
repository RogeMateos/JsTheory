Promesas en JavaScript

¿Qué es una promesa?

Objeto que representa un resultado futuro: pendiente, resuelta o rechazada. Como pedir comida a domicilio: tienes la promesa, pero aún no llega.

Estados

  1. Pendiente
  2. Resuelta
  3. Rechazada

Crear y consumir

const myPromise = new Promise((resolve, reject) => {
  setTimeout(() => {
    const ok = true;
    ok ? resolve("Listo") : reject("Error");
  }, 1000);
});

myPromise
  .then(res => console.log(res))
  .catch(err => console.error(err))
  .finally(() => console.log("Siempre se ejecuta"));

Ejemplo práctico: pizza 🍕

function orderPizza(type) {
  return new Promise((resolve, reject) => {
    console.log(`Preparando ${type}...`);
    setTimeout(() => {
      const menu = ['pepperoni', 'margherita'];
      menu.includes(type)
        ? resolve(`Tu ${type} está lista 🍕`)
        : reject(`No tenemos ${type}`);
    }, 1500);
  });
}

orderPizza('pepperoni')
  .then(msg => console.log(msg))
  .catch(err => console.error(err));

Encadenar promesas

function step(msg, t) {
  return new Promise(res => setTimeout(() => {
    console.log(msg);
    res();
  }, t));
}

step("Paso 1", 500)
  .then(() => step("Paso 2", 500))
  .then(() => step("Paso 3", 500))
  .then(() => console.log("Completado 🎉"));

Vs callback hell

// Antes
operation1(function(r1) {
  operation2(r1, function(r2) {
    operation3(r2, function(r3) {
      operation4(r3, function(r4) {
        // difícil de leer
      });
    });
  });
});

// Con promesas
operation1()
  .then(r1 => operation2(r1))
  .then(r2 => operation3(r2))
  .then(r3 => operation4(r3))
  .catch(console.error);

Lado A/Lado B

resolve = éxito, reject = error. Maneja ambos caminos siempre.
