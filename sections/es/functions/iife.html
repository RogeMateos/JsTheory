IIFE - Immediately Invoked Function Expression

Qué son las IIFEs

IIFE (Immediately Invoked Function Expression) es una función que se ejecuta en el mismo momento en que se define. No necesitas llamarla explícitamente después.


// Función Normal - Primero la defines, luego la llamas
function saludar() {
  console.log('Hola desde función normal');
}
saludar(); // Output: "Hola desde función normal"

// IIFE - Se define Y se ejecuta al mismo tiempo
(function() {
  console.log('Hola desde IIFE');
})();
// Output: "Hola desde IIFE"
// No necesitaste llamarla por nombre, se ejecutó sola

Sintaxis y Variantes

Anatomía de una IIFE


// PASO 1: Una función anónima normal
function() {
  console.log('Esto es una función');
}
// ❌ Esto da error: Function statements require a function name

// PASO 2: Convertirla en expresión con paréntesis
(function() {
  console.log('Ahora es una expresión de función');
})
// ✅ Ya no da error, pero tampoco se ejecuta

// PASO 3: Invocarla inmediatamente
(function() {
  console.log('¡Ahora se ejecuta!');
})();
// ✅ Output: "¡Ahora se ejecuta!"

/*
ESTRUCTURA:
  (function() {    ← Paréntesis convierte en expresión
    // código
  })               ← Cierra la expresión
  ();              ← Invoca la función inmediatamente
*/

IIFE con Arrow Function


// Sintaxis con arrow function (ES6+)
(() => {
  console.log('IIFE con arrow function');
})();
// Output: "IIFE con arrow function"

// Versión compacta (una sola expresión)
(() => console.log('IIFE compacta'))();
// Output: "IIFE compacta"

Dos Formas Válidas de Paréntesis


// FORMA 1: Paréntesis fuera (MÁS COMÚN)
(function() {
  console.log('Paréntesis externos');
})();

// FORMA 2: Paréntesis dentro (MENOS COMÚN)
(function() {
  console.log('Paréntesis internos');
}());

// Ambas funcionan exactamente igual
// La primera es más popular en la comunidad

El Problema del Punto y Coma

JavaScript intenta insertar punto y coma automáticamente (ASI - Automatic Semicolon Insertion), pero a veces falla con IIFEs.


// ❌ EL PROBLEMA
const saludo = 'Hola'

(function() {
  console.log('IIFE');
})();

/*
Error: TypeError: "Hola" is not a function

¿QUÉ PASÓ?
JavaScript interpreta esto como:
const saludo = 'Hola'(function() { ... })();
Intenta llamar a 'Hola' como función!
*/

// ✅ SOLUCIÓN 1: Punto y coma ANTES de la IIFE
const saludo2 = 'Hola';  // ← Punto y coma aquí

(function() {
  console.log('IIFE correcta');
})();

// ✅ SOLUCIÓN 2: Punto y coma DEFENSIVO al inicio
const mensaje = 'Mundo'

;(function() {  // ← Punto y coma defensivo
  console.log('IIFE con ; defensivo');
})();

// 🎯 REGLA PRÁCTICA:
// Si no usas punto y coma habitualmente:
// → Siempre inicia IIFEs con punto y coma defensivo
// ;(function() { ... })();

IIFEs con Parámetros


// Pasar argumentos a una IIFE
(function(nombre) {
  console.log('Hola,', nombre);
})('Roge');
// Output: "Hola, Roge"

// Múltiples parámetros
(function(nombre, edad, ciudad) {
  console.log(`${nombre} tiene ${edad} años y vive en ${ciudad}`);
})('Ana', 28, 'Madrid');
// Output: "Ana tiene 28 años y vive en Madrid"

// IIFE que retorna un valor
const resultado = (function(a, b) {
  return a * b;
})(4, 5);

console.log('Resultado:', resultado);
// Output: "Resultado: 20"
// IMPORTANTE: resultado NO es una función, es el valor 20

Casos de Uso Prácticos

1. Encapsulación de Variables


// ❌ SIN IIFE - Contamina el scope global
var contador = 0;
var apiKey = 'secreto-123';

// Estas variables están expuestas globalmente
console.log(window.contador);  // 0 (accesible globalmente)

// ✅ CON IIFE - Variables encapsuladas
(function() {
  var contador = 0;
  var apiKey = 'secreto-123';

  function incrementar() {
    contador++;
    console.log('Contador:', contador);
  }

  incrementar();  // Output: "Contador: 1"
})();

// console.log(contador);  // ❌ Error: contador is not defined

2. Module Pattern (Patrón Módulo)


const AppModule = (function() {
  // Variables privadas
  let usuarios = [];
  const API_KEY = 'abc123';

  // Funciones privadas
  function validarUsuario(nombre) {
    return nombre && nombre.length > 0;
  }

  // API pública (lo que se exporta)
  return {
    agregarUsuario(nombre) {
      if (validarUsuario(nombre)) {
        usuarios.push(nombre);
        console.log(`Usuario ${nombre} agregado`);
        return true;
      }
      return false;
    },

    obtenerUsuarios() {
      return [...usuarios];  // Retorna copia
    },

    cantidadUsuarios() {
      return usuarios.length;
    }
  };
})();

// Uso del módulo
AppModule.agregarUsuario('Ana');     // Output: "Usuario Ana agregado"
console.log(AppModule.obtenerUsuarios());  // Output: ['Ana']
// console.log(AppModule.API_KEY);   // ❌ undefined (privado)

3. Async/Await en IIFE


// ✅ IIFE async para ejecutar código async inmediatamente
(async function() {
  try {
    console.log('Cargando datos...');

    const respuesta = await fetch('https://api.ejemplo.com/users/1');
    const usuario = await respuesta.json();

    console.log('Usuario cargado:', usuario.name);

  } catch (error) {
    console.error('Error:', error);
  }
})();

// Con arrow function
(async () => {
  const data = await fetch('https://api.ejemplo.com/posts/1');
  const post = await data.json();
  console.log('Post:', post.title);
})();

4. Contador Privado


const contador = (function() {
  let cuenta = 0;  // Variable privada

  return function() {
    cuenta++;
    return cuenta;
  };
})();

console.log(contador());  // Output: 1
console.log(contador());  // Output: 2
console.log(contador());  // Output: 3
// console.log(cuenta);   // ❌ Error: cuenta is not defined

Errores Comunes


// ❌ ERROR 1: Olvidar paréntesis externos
function() {
  console.log('Error');
}();
// SyntaxError: Function statements require a function name

// ✅ CORRECTO:
(function() {
  console.log('Correcto');
})();

// ❌ ERROR 2: Olvidar paréntesis de invocación
(function() {
  console.log('Esto nunca se ejecuta');
});
// No hay error, pero la función NUNCA se ejecuta

// ✅ CORRECTO:
(function() {
  console.log('Ahora sí se ejecuta');
})();  // ← Los () son cruciales

// ❌ ERROR 3: Pasar parámetros incorrectamente
(function(nombre) {
  console.log('Hola', nombre);  // undefined
})();
// Faltó pasar el argumento

// ✅ CORRECTO:
(function(nombre) {
  console.log('Hola', nombre);
})('Roge');  // Output: "Hola Roge"

JavaScript Moderno vs IIFEs

Antes vs Ahora


// ❌ ANTES (Pre-ES6): IIFEs para encapsulación
(function() {
  var configuracion = { apiUrl: 'https://api.ejemplo.com' };
  var apiKey = 'secreto123';

  function inicializar() {
    console.log('App inicializada');
  }

  window.MiApp = { inicializar };
})();

// ✅ AHORA (ES6+): Módulos nativos
// archivo: config.js
export const configuracion = { apiUrl: 'https://api.ejemplo.com' };
const apiKey = 'secreto123';  // Privado (no exportado)

export function inicializar() {
  console.log('App inicializada');
}

// archivo: app.js
import { configuracion, inicializar } from './config.js';

Block Scope con let/const


// ❌ ANTES: IIFE para crear scope
(function() {
  var temporal = 'Solo existe aquí';
  console.log(temporal);
})();

// ✅ AHORA: Block scope con {}
{
  const temporal = 'Solo existe aquí';
  console.log(temporal);
}
// Más simple, menos sintaxis

Cuándo TODAVÍA Usar IIFEs

Las IIFEs siguen siendo útiles en estos casos:

  • Scripts inline en HTML: En tags <script> sin module type
  • Browser console: Al escribir código directo en consola
  • Bookmarklets: Pequeños scripts en marcadores
  • Código legacy: Cuando trabajas con código antiguo sin build tools
  • Polyfills y bibliotecas: Para compatibilidad

// Ejemplo: Script inline en HTML
<script>
  (function() {
    // Evita contaminar scope global
    const miVariable = 'privada';
    console.log(miVariable);
  })();
</script>

// Ejemplo: Inicialización única de config
const CONFIG = (function() {
  const env = process.env.NODE_ENV || 'development';

  return env === 'development'
    ? { apiUrl: 'http://localhost:3000', debug: true }
    : { apiUrl: 'https://api.produccion.com', debug: false };
})();

Resumen

Puntos Clave:

  • Definición: Immediately Invoked Function Expression - función que se ejecuta al definirse
  • Sintaxis: (function() { ... })() o (() => { ... })()
  • Usos: Encapsulación, evitar contaminación del scope global, crear módulos privados
  • Punto y coma: Usar ; defensivo al inicio si no usas ; habitualmente
  • Moderno: Módulos ES6 reemplazan muchos usos, pero IIFEs siguen siendo útiles
